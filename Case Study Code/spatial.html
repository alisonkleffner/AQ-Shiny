#Create Interactive Plot
plot_ly(
  data = spatial_data,
  lon = ~Longitude,
  lat = ~Latitude,
  type = 'scattermapbox',
  mode = 'markers',
  color = ~PM2.5, # Color by a categorical variable, e.g., status
  colors = viridis::cividis(n = 100), #color blind friendly palette
  marker = list(size = 10, opacity = 0.8),
  showlegend = FALSE,
  text = ~paste("PM2.5: ", round(PM2.5, 4)),
  hoverinfo = "text"
  ) %>%
  layout(
    mapbox = list(
    style = "open-street-map", #zoom into california
    zoom = 5,
    center = list(
    lon = mean(spatial_data$Longitude),
    lat = mean(spatial_data$Latitude)
    )
  )
)

MODELING TAB

# Exponential Sphere Covariance Function  

locs <- as.matrix(spatial_data[, c("Longitude", "Latitude")])
response  <- spatial_data$PM2.5

X <- matrix(1, nrow(locs), 1) # Intercept-only mean model

fit_exp <- fit_model(
  y = response,
  locs = locs,
  X = X,
  covfun_name = "exponential_sphere", 
  m_seq = c(15, 30), max_iter = 100,
  convtol = 1e-05, reorder = TRUE,
  silent = TRUE)

# Read in Prediction Data

# Exponential Sphere Covariance Function  

longlat_pred<-spatial_pred[,c("Longitude","Latitude")]

X_pred <- as.matrix(rep(1,nrow(locs_pred)))

prediction_exp <- predictions(fit_exp,locs_pred, X_pred, covparms = fit_exp$covparms, covfun_name = fit_exp$covfun_name, y_obs = fit_exp$y, locs_obs = fit_exp$locs, X_obs= fit_exp$X, beta=fit_exp$betahat, m = 60, reorder = TRUE)
